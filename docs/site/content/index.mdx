PPTX Editor is a WYSIWYG editor for `.pptx` presentations in React and Vue 3. It reads the package, lays out each slide by the ECMA-376 rules, paints it as SVG, and saves the edited deck in the browser. Editing needs no upload or conversion service.

## Install an editor

Install an adapter and the engine:

<FrameworkTabs>
<Framework value="react">

```bash
npm install @pptx-editor.dev/react @pptx-editor.dev/core
```

</Framework>
<Framework value="vue">

```bash
npm install @pptx-editor.dev/vue @pptx-editor.dev/core
```

</Framework>
</FrameworkTabs>

Render the editor in a container with a height:

<FrameworkTabs>
<Framework value="react">

```tsx
import { PptxEditor } from '@pptx-editor.dev/react';
import '@pptx-editor.dev/core/styles/editor.css';

export default function App({ bytes }: { bytes: Uint8Array }) {
  return (
    <div style={{ height: '100vh' }}>
      <PptxEditor document={bytes} mode="edit" />
    </div>
  );
}
```

</Framework>
<Framework value="vue">

```vue
<script setup lang="ts">
import { PptxEditor } from '@pptx-editor.dev/vue';
import '@pptx-editor.dev/core/styles/editor.css';

defineProps<{ bytes: Uint8Array }>();
</script>

<template>
  <div style="height: 100vh">
    <PptxEditor :document="bytes" mode="edit" />
  </div>
</template>
```

</Framework>
</FrameworkTabs>

To open and save files, continue with the [quickstart](/docs/quickstart).
